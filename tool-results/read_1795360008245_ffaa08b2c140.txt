   375→   375→      idx.push(Math.min(len - 1, Math.floor(((i + 0.5) * len) / n)));

   378→   378→  }, [count, gifAll, gifFrames.length, mode]);
   379→   379→
   380→   380→  const extract = useCallback(async () => {
   381→   381→    if (mode === "idle") {
   382→   382→      toast.error("ابتدا ویدیو یا گیف بارگذاری کنید");

   390→   390→    const chroma = chromaRef.current;
   391→   391→    const applyTo = (canvas: HTMLCanvasElement): HTMLCanvasElement => {
   392→   392→      if (!chroma.enabled) return canvas;
   393→   393→      const ctx = canvas.getContext("2d", { willReadFrequently: true })!;
   394→   394→      const img = ctx.getImageData(0, 0, canvas.width, canvas.height);
   395→   395→      applyChromaKey(img, {
   396→   396→        color: hexToRgb(chroma.hex),
   397→   397→        similarity: chroma.similarity,
   398→   398→        smoothness: chroma.smoothness,
   399→   399→        despill: chroma.despill,
   400→   400→      });
   401→   401→      ctx.putImageData(img, 0, 0);
   402→   402→      return canvas;
   403→   403→    };
   404→   404→
   405→   405→    try {
   406→   406→      const inputs: Array<{ name: string; canvas: HTMLCanvasElement; source: "video" }> = [];
   407→   407→
   408→   408→      if (mode === "video") {
   409→   409→        const v = videoElRef.current;
   410→   410→        if (!v || !videoMeta) return;
   411→   411→        const frames = await extractFramesFromVideo(
   412→   412→          v,
   413→   413→          videoMeta.duration,
   414→   414→          count,
   415→   415→          (p) => setProgress(p),
   416→   416→          () => cancelRef.current
   417→   417→        );
   418→   418→        frames.forEach((f, i) => {
   419→   419→          inputs.push({
   420→   420→            name: frameName(`${prefix.trim() || "vid"}_{i}`, i, frames.length),
   421→   421→            canvas: applyTo(f.canvas),
   422→   422→            source: "video",
   423→   423→          });
   424→   424→        });
   425→   425→      } else {
   426→   426→        const idx = targetIndices();
   427→   427→        const list = Array.isArray(idx) ? idx : [];
   428→   428→        for (let i = 0; i < list.length; i++) {
   429→   429→          if (cancelRef.current) break;
   430→   430→          const f = gifFrames[list[i]];
   431→   431→          if (!f) continue;
   432→   432→          const canvas = applyTo(cloneCanvas(f.canvas));
   433→   433→          inputs.push({
   434→   434→            name: frameName(`${prefix.trim() || "vid"}_{i}`, i, list.length),
   435→   435→            canvas,
   436→   436→            source: "video",
   437→   437→          });
   438→   438→          setProgress({ current: i + 1, total: list.length });
   439→   439→          await new Promise((r) => setTimeout(r, 0));
   440→   440→        }
   441→   441→      }
   442→   442→
   443→   443→      if (inputs.length === 0) {
