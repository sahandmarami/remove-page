   108→export function SheetCutter() {

   110→  const addFrame = useFrameStore((s) => s.addFrame);
   111→  const fileInputRef = useRef<HTMLInputElement>(null);
   112→
   113→  const [source, setSource] = useState<SourceImage | null>(null);
   114→  const [boxes, setBoxes] = useState<Box[]>([]);
   115→  const [selectedId, setSelectedId] = useState<string | null>(null);
   116→  const [detectTolerance, setDetectTolerance] = useState(42);
   117→  const [aiBusy, setAiBusy] = useState(false);
   118→  const [localBusy, setLocalBusy] = useState(false);
   119→
   120→  const [bgEnabled, setBgEnabled] = useState(true);
   121→  const [bgTolerance, setBgTolerance] = useState(DEFAULT_BG_OPTIONS.tolerance);
   122→  const [edgeBarrier, setEdgeBarrier] = useState(DEFAULT_BG_OPTIONS.edgeBarrier);
   123→  const [feather, setFeather] = useState(DEFAULT_BG_OPTIONS.feather);
   124→  const [defringe, setDefringe] = useState(DEFAULT_BG_OPTIONS.defringe);
   125→  const [removeHoles, setRemoveHoles] = useState(DEFAULT_BG_OPTIONS.removeHoles);
   126→  const [holeCap, setHoleCap] = useState(DEFAULT_BG_OPTIONS.holeMaxPercent);
   127→  const [bgSwatch, setBgSwatch] = useState<RGB | null>(null);
   128→
   129→  const [prefix, setPrefix] = useState("sprite");
   130→  const [autoDownload, setAutoDownload] = useState(true);
   131→  const [cropping, setCropping] = useState(false);
   132→  const [maskKey, setMaskKey] = useState(0);
   133→  const [maskState, setMaskState] = useState<{
   134→    original: HTMLCanvasElement;
   135→    processed: HTMLCanvasElement;
   136→    name: string;
   137→  } | null>(null);
   138→
   139→  const loadCanvas = useCallback((canvas: HTMLCanvasElement, name: string) => {
   140→    setSource({ canvas, name, w: canvas.width, h: canvas.height });
   141→    setBoxes([]);
   142→    setSelectedId(null);
   143→    setBgSwatch(null);
   144→    toast.success("تصویر بارگذاری شد", { description: `${name} — ${canvas.width}×${canvas.height}` });
   145→  }, []);
   146→
   147→  const onFiles = useCallback(
   148→    async (files: FileList | null) => {
   149→      const file = files?.[0];
   150→      if (!file) return;
   151→      if (!file.type.startsWith("image/")) {
   152→        toast.error("فقط فایل تصویری پذیرفته می‌شود");
   153→        return;
   154→      }
   155→      try {
   156→        const { canvas } = await loadImageToCanvas(file);
   157→        loadCanvas(canvas, file.name);
   158→      } catch {
   159→        toast.error("بارگذاری تصویر ناموفق بود");
   160→      }
   161→    },
   162→    [loadCanvas]
   163→  );
   164→
   165→  const loadSample = useCallback(() => {
   166→    const canvas = generateSampleSheet();
   167→    loadCanvas(canvas, "sample-sheet.png");
   168→  }, [loadCanvas]);
   169→
   170→  const localDetect = useCallback(() => {
   171→    if (!source) {
   172→      toast.error("ابتدا یک تصویر بارگذاری کنید");
   173→      return;
   174→    }
   175→    setLocalBusy(true);
   176→    // let the spinner paint before the synchronous CV pass
   177→    window.setTimeout(() => {
   178→      try {
   179→        const result = detectBoxes(source.canvas, { tolerance: detectTolerance });
   180→        setBoxes(result.boxes);
   181→        setSelectedId(null);
   182→        setBgSwatch(result.bgColor);
   183→        if (result.boxes.length === 0) {
   184→          toast.warning("آیتمی پیدا نشد", {
   185→            description: "آستانه حساسیت را کاهش دهید یا از تشخیص هوش مصنوعی استفاده کنید",
   186→          });
   187→        } else {
   188→          toast.success(`${result.boxes.length} آیتم شناسایی شد`, {
   189→            description: `رنگ پس‌زمینه: ${rgbToHex(result.bgColor)}`,
   190→          });
   191→        }
   192→      } catch (err) {
   193→        toast.error(err instanceof Error ? err.message : "خطا در تشخیص محلی");
   194→      } finally {
   195→        setLocalBusy(false);
   196→      }
   197→    }, 40);
   198→  }, [detectTolerance, source]);
   199→
   200→  const aiDetect = useCallback(async () => {
   201→    if (!source) {
   202→      toast.error("ابتدا یک تصویر بارگذاری کنید");
   203→      return;
   204→    }
   205→    setAiBusy(true);
   206→    try {
   207→      // downscale for a light payload
   208→      const maxSide = 1024;
   209→      const scale = Math.min(1, maxSide / Math.max(source.w, source.h));
   210→      const small = document.createElement("canvas");
   211→      small.width = Math.round(source.w * scale);
   212→      small.height = Math.round(source.h * scale);
   213→      small.getContext("2d")!.drawImage(source.canvas, 0, 0, small.width, small.height);
   214→      const dataUrl = small.toDataURL("image/jpeg", 0.85);
   215→
   216→      const controller = new AbortController();
   217→      const timer = window.setTimeout(() => controller.abort(), 100_000);
   218→      const res = await fetch("/api/detect", {
   219→        method: "POST",
   220→        headers: { "Content-Type": "application/json" },
   221→        body: JSON.stringify({ image: dataUrl, width: source.w, height: source.h }),
   222→        signal: controller.signal,
   223→      });
   224→      window.clearTimeout(timer);
   225→      const json = (await res.json()) as {
   226→        ok: boolean;
   227→        boxes?: Array<{ x: number; y: number; w: number; h: number; label?: string }>;
   228→        error?: string;
   229→      };
   230→      if (!res.ok || !json.ok) {
   231→        throw new Error(json.error || `پاسخ سرور: ${res.status}`);
   232→      }
   233→      const mapped: Box[] = (json.boxes ?? []).map((b) => ({
   234→        id: makeBoxId(),
   235→        x: Math.round(b.x * source.w),
   236→        y: Math.round(b.y * source.h),
   237→        w: Math.round(b.w * source.w),
   238→        h: Math.round(b.h * source.h),
   239→        label: b.label,
   240→      }));
   241→      setBoxes(mapped);
   242→      setSelectedId(null);
   243→      if (mapped.length === 0) {
   244→        toast.warning("هوش مصنوعی کادری برنگرداند", {
   245→          description: "تشخیص محلی بینایی ماشین را امتحان کنید",
   246→        });
   247→      } else {
   248→        toast.success(`${mapped.length} کادر از Vision AI دریافت شد`);
   249→      }
   250→    } catch (err) {
   251→      const msg =
   252→        err instanceof DOMException && err.name === "AbortError"
   253→          ? "زمان پاسخ سرور به پایان رسید"
   254→          : err instanceof Error
   255→            ? err.message
   256→            : "خطای ناشناخته";
   257→      toast.error("تشخیص هوش مصنوعی ناموفق بود", { description: msg });
   258→    } finally {
   259→      setAiBusy(false);
   260→    }
   261→  }, [source]);
   262→
   263→  const processCrop = useCallback(
   264→    (box: Box): { original: HTMLCanvasElement; processed: HTMLCanvasElement } => {
   265→      const original = cropToCanvas(source!.canvas, box);
   266→      if (!bgEnabled) {
   267→        return { original, processed: original };
   268→      }
   269→      const img = original.getContext("2d", { willReadFrequently: true })!.getImageData(
   270→        0,
   271→        0,
   272→        original.width,
   273→        original.height
   274→      );
   275→      const result = removeBackground(img, {
   276→        tolerance: bgTolerance,
   277→        edgeBarrier,
   278→        feather,
   279→        defringe,
   280→        removeHoles,
   281→        holeMaxPercent: holeCap,
   282→      });
   283→      const processed = document.createElement("canvas");
   284→      processed.width = original.width;
   285→      processed.height = original.height;
   286→      processed.getContext("2d")!.putImageData(result.image, 0, 0);
   287→      return { original, processed };
   288→    },
   289→    [bgEnabled, bgTolerance, defringe, edgeBarrier, feather, holeCap, removeHoles, source]
   290→  );
   291→
   292→  const openMaskEditor = useCallback(() => {
   293→    if (!source) return;
   294→    const box = boxes.find((b) => b.id === selectedId);
   295→    if (!box) {
   296→      toast.error("یک کادر را انتخاب کنید");
   297→      return;
   298→    }
   299→    const { original, processed } = processCrop(box);
   300→    const idx = boxes.findIndex((b) => b.id === selectedId);
   301→    setMaskKey((k) => k + 1);
   302→    setMaskState({
   303→      original,
   304→      processed,
   305→      name: frameName(`${prefix.trim() || "sprite"}_{i}`, idx, boxes.length),
   306→    });
   307→  }, [boxes, prefix, processCrop, selectedId, source]);
   308→
   309→  /** cut every box → add to animator → optionally download a transparent-WebP ZIP */
   310→  const cropAll = useCallback(async () => {
   311→    if (!source) return;
   312→    if (boxes.length === 0) {
   313→      toast.error("کادری برای برش وجود ندارد");
   314→      return;
   315→    }
   316→    setCropping(true);
   317→    try {
   318→      const inputs: Array<{ name: string; canvas: HTMLCanvasElement; source: "sheet" }> = [];
   319→      for (let i = 0; i < boxes.length; i++) {
   320→        const { processed } = processCrop(boxes[i]);
   321→        inputs.push({
   322→          name: frameName(`${prefix.trim() || "sprite"}_{i}`, i, boxes.length),
   323→          canvas: processed,
   324→          source: "sheet",
   325→        });
   326→        // keep UI responsive
   327→        await new Promise((r) => setTimeout(r, 0));
   328→      }
   329→      const n = addFrames(inputs);
   330→      if (autoDownload) {
   331→        toast.loading("در حال ساخت ZIP شفاف...", { id: "crop-zip" });
   332→        await downloadFramesZip(inputs, `${prefix.trim() || "sprite"}-cut`, 0.95);
   333→        toast.success(`${n} فایل برش‌شده دانلود شد`, {
   334→          id: "crop-zip",
   335→          description: bgEnabled
   336→            ? "ZIP با فریم‌های WebP شفاف — در انیماتور هم آماده است"
   337→            : "در تب انیماتور هم آماده است",
   338→        });
   339→      } else {
   340→        toast.success(`${n} فریم به انیماتور اضافه شد`, {
   341→          description: bgEnabled ? "با حذف پس‌زمینه و لبه نرم" : "برش مستقیم بدون حذف پس‌زمینه",
   342→        });
   343→      }
   344→    } catch (err) {
   345→      toast.error(err instanceof Error ? err.message : "خطا در برش فریم‌ها");
   346→    } finally {
   347→      setCropping(false);
   348→    }
   349→  }, [addFrames, autoDownload, bgEnabled, boxes, prefix, processCrop, source]);
   350→
   351→  /** download a single cut frame as transparent WebP */
   352→  const downloadOneBox = useCallback(
   353→    async (box: Box, index: number) => {
   354→      if (!source) return;
   355→      try {
   356→        const { processed } = processCrop(box);
   357→        const name = frameName(`${prefix.trim() || "sprite"}_{i}`, index, boxes.length);
   358→        const { blob, actualType } = await canvasToBlob(processed, "image/webp", 0.95);
   359→        downloadBlob(blob, `${name}.${extFor(actualType)}`);
   360→        toast.success(`«${name}.webp» دانلود شد`);
   361→      } catch (err) {
   362→        toast.error(err instanceof Error ? err.message : "دانلود فریم ناموفق بود");
   363→      }
   364→    },
   365→    [boxes.length, prefix, processCrop, source]
   366→  );
   367→
   368→  // graphical thumbnails for detected boxes (recomputed only when boxes settle)
   369→  const boxThumbs = useMemo(() => {
   370→    if (!source) return [];
   371→    return boxes.map((b, i) => {
   372→      const crop = cropToCanvas(source.canvas, b);
   373→      return {
   374→        id: b.id,
   375→        index: i,
   376→        thumb: makeThumb(crop, 132),
   377→        title: `${b.label ? `${b.label} — ` : ""}${Math.round(b.w)}×${Math.round(b.h)}`,
   378→      };
   379→    });
   380→  }, [boxes, source]);
   381→
   382→  const selectedBox = boxes.find((b) => b.id === selectedId) ?? null;
   383→
   384→  // ---------- empty state: graphical dropzone hero ----------
   385→  if (!source) {
   386→    return (
   387→      <div className="flex flex-col gap-4">
   388→        <input
   389→          ref={fileInputRef}
   390→          type="file"
   391→          accept="image/*"
   392→          className="hidden"
   393→          onChange={(e) => void onFiles(e.target.files)}
   394→        />
   395→        <div
   396→          className="hero-dots relative cursor-pointer overflow-hidden rounded-3xl border-2 border-dashed border-violet-400/25 bg-slate-950/40 p-8 text-center transition-colors hover:border-violet-400/50 sm:p-12"
   397→          onClick={() => fileInputRef.current?.click()}
   398→          onDragOver={(e) => e.preventDefault()}
   399→          onDrop={(e) => {
   400→            e.preventDefault();
   401→            void onFiles(e.dataTransfer.files);
   402→          }}
   403→        >
   404→          <div className="pointer-events-none absolute -top-28 right-1/2 h-72 w-72 translate-x-1/2 rounded-full bg-violet-500/20 blur-3xl" />
   405→          <div className="pointer-events-none absolute -bottom-28 left-1/4 h-56 w-56 rounded-full bg-cyan-400/10 blur-3xl" />
   406→          <div className="relative">
   407→            <div className="hero-icon-glow mx-auto grid h-20 w-20 place-items-center rounded-2xl metallic-gradient">
   408→              <ImageUp className="h-9 w-9 text-white" />
   409→            </div>
   410→            <h2 className="mt-5 text-xl font-black metallic-text sm:text-2xl">
   411→              شیت تصویر را اینجا رها کنید
   412→            </h2>
   413→            <p className="mx-auto mt-2 max-w-xl text-sm leading-6 text-slate-400">
   414→              آیتم‌ها خودکار شناسایی می‌شوند، پس‌زمینه حذف می‌شود و با یک کلیک، همه فایل‌های
   415→              برش‌خورده به‌صورت ZIP شفاف دانلود می‌شوند.
   416→            </p>
   417→            <div className="mt-6 flex flex-wrap items-center justify-center gap-3">
   418→              <Button
   419→                type="button"
   420→                onClick={(e) => {
   421→                  e.stopPropagation();
   422→                  fileInputRef.current?.click();
   423→                }}
   424→                className="metallic-gradient text-white font-bold glow-purple hover:brightness-110"
   425→              >
   426→                <ImageUp className="h-4 w-4" />
   427→                انتخاب تصویر
   428→              </Button>
   429→              <Button
   430→                type="button"
   431→                variant="secondary"
   432→                onClick={(e) => {
   433→                  e.stopPropagation();
   434→                  loadSample();
   435→                }}
   436→                className="border border-white/10 bg-white/5 hover:border-cyan-400/40"
   437→              >
   438→                <Sparkles className="h-4 w-4 text-cyan-400" />
   439→                شیت نمونه تستی
   440→              </Button>
   441→            </div>
   442→            <p className="mt-3 text-[11px] text-slate-500" dir="ltr">
   443→              PNG · JPG · WebP — Drag &amp; Drop
   444→            </p>
   445→
   446→            <div className="mx-auto mt-9 grid max-w-3xl gap-3 sm:grid-cols-3">
   447→              {FEATURES.map((f) => (
   448→                <div
   449→                  key={f.title}
   450→                  className="rounded-2xl border border-white/10 bg-slate-950/60 p-4 text-right transition-colors hover:border-white/25"
   451→                >
   452→                  <span className={cn("mb-2.5 grid h-9 w-9 place-items-center rounded-xl border", f.cls)}>
   453→                    <f.icon className="h-4.5 w-4.5" />
   454→                  </span>
   455→                  <p className="text-sm font-bold text-slate-200">{f.title}</p>
   456→                  <p className="mt-1 text-[11px] leading-4 text-slate-500">{f.desc}</p>
   457→                </div>

   459→            </div>
   460→          </div>

   462→      </div>

   464→  }
   465→
   466→  // ---------- loaded state ----------
   467→  return (
   468→    <div className="flex flex-col gap-4">
   469→      <input
   470→        ref={fileInputRef}
   471→        type="file"
   472→        accept="image/*"
   473→        className="hidden"
   474→        onChange={(e) => void onFiles(e.target.files)}
   475→      />
   476→
   477→      <div className="grid gap-4 xl:grid-cols-[1fr_340px]">
   478→        {/* ---------- canvas column ---------- */}
   479→        <div className="flex flex-col gap-3">
   480→          {/* compact toolbar */}
   481→          <div className="flex flex-wrap items-center gap-2 rounded-2xl panel-glass p-2.5">
   482→            <Button
   483→              type="button"
   484→              size="sm"
   485→              variant="secondary"
   486→              onClick={() => fileInputRef.current?.click()}

   503→            <Button
   504→              type="button"

   506→              variant="secondary"
   507→              disabled={localBusy}

   509→              className="border border-cyan-400/30 bg-cyan-400/10 text-cyan-200 hover:border-cyan-400/60"

   554→          <div
