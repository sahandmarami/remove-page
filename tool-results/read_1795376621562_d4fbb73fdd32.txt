   108→export function SheetCutter() {
   109→  const addFrames = useFrameStore((s) => s.addFrames);
   110→  const addFrame = useFrameStore((s) => s.addFrame);
   111→  const fileInputRef = useRef<HTMLInputElement>(null);
   112→
   113→  const [source, setSource] = useState<SourceImage | null>(null);
   114→  const [boxes, setBoxes] = useState<Box[]>([]);
   115→  const [selectedId, setSelectedId] = useState<string | null>(null);
   116→  const [detectTolerance, setDetectTolerance] = useState(42);
   117→  const [aiBusy, setAiBusy] = useState(false);
   118→  const [localBusy, setLocalBusy] = useState(false);
   119→
   120→  const [bgEnabled, setBgEnabled] = useState(true);
   121→  const [bgTolerance, setBgTolerance] = useState(DEFAULT_BG_OPTIONS.tolerance);
   122→  const [edgeBarrier, setEdgeBarrier] = useState(DEFAULT_BG_OPTIONS.edgeBarrier);
   123→  const [feather, setFeather] = useState(DEFAULT_BG_OPTIONS.feather);
   124→  const [defringe, setDefringe] = useState(DEFAULT_BG_OPTIONS.defringe);
   125→  const [removeHoles, setRemoveHoles] = useState(DEFAULT_BG_OPTIONS.removeHoles);
   126→  const [holeCap, setHoleCap] = useState(DEFAULT_BG_OPTIONS.holeMaxPercent);
   127→  const [bgSwatch, setBgSwatch] = useState<RGB | null>(null);
   128→
   129→  const [prefix, setPrefix] = useState("sprite");
   130→  const [autoDownload, setAutoDownload] = useState(true);
   131→  const [cropping, setCropping] = useState(false);
   132→  const [maskKey, setMaskKey] = useState(0);
   133→  const [maskState, setMaskState] = useState<{
   134→    original: HTMLCanvasElement;
   135→    processed: HTMLCanvasElement;
   136→    name: string;
   137→  } | null>(null);
   138→
   139→  const loadCanvas = useCallback((canvas: HTMLCanvasElement, name: string) => {
   140→    setSource({ canvas, name, w: canvas.width, h: canvas.height });
   141→    setBoxes([]);
   142→    setSelectedId(null);
   143→    setBgSwatch(null);
   144→    toast.success("تصویر بارگذاری شد", { description: `${name} — ${canvas.width}×${canvas.height}` });
   145→  }, []);
   146→
   147→  const onFiles = useCallback(
   148→    async (files: FileList | null) => {
   149→      const file = files?.[0];
   150→      if (!file) return;
   151→      if (!file.type.startsWith("image/")) {
   152→        toast.error("فقط فایل تصویری پذیرفته می‌شود");
   153→        return;
   154→      }
   155→      try {
   156→        const { canvas } = await loadImageToCanvas(file);
   157→        loadCanvas(canvas, file.name);
   158→      } catch {
   159→        toast.error("بارگذاری تصویر ناموفق بود");
   160→      }
   161→    },
   162→    [loadCanvas]
   163→  );
   164→
   165→  const loadSample = useCallback(() => {
   166→    const canvas = generateSampleSheet();
   167→    loadCanvas(canvas, "sample-sheet.png");
   168→  }, [loadCanvas]);
   169→
   170→  const localDetect = useCallback(() => {
   171→    if (!source) {
   172→      toast.error("ابتدا یک تصویر بارگذاری کنید");
   173→      return;
   174→    }
   175→    setLocalBusy(true);
   176→    // let the spinner paint before the synchronous CV pass
   177→    window.setTimeout(() => {
   178→      try {
   179→        const result = detectBoxes(source.canvas, { tolerance: detectTolerance });
   180→        setBoxes(result.boxes);
   181→        setSelectedId(null);
   182→        setBgSwatch(result.bgColor);
   183→        if (result.boxes.length === 0) {
   184→          toast.warning("آیتمی پیدا نشد", {
   185→            description: "آستانه حساسیت را کاهش دهید یا از تشخیص هوش مصنوعی استفاده کنید",
   186→          });
   187→        } else {
   188→          toast.success(`${result.boxes.length} آیتم شناسایی شد`, {
   189→            description: `رنگ پس‌زمینه: ${rgbToHex(result.bgColor)}`,
   190→          });
   191→        }
   192→      } catch (err) {
   193→        toast.error(err instanceof Error ? err.message : "خطا در تشخیص محلی");
   194→      } finally {
   195→        setLocalBusy(false);
   196→      }
   197→    }, 40);
   198→  }, [detectTolerance, source]);
   199→
   200→  const aiDetect = useCallback(async () => {
   201→    if (!source) {
   202→      toast.error("ابتدا یک تصویر بارگذاری کنید");
   203→      return;
   204→    }
   205→    setAiBusy(true);
   206→    try {
   207→      // downscale for a light payload
   208→      const maxSide = 1024;
   209→      const scale = Math.min(1, maxSide / Math.max(source.w, source.h));
   210→      const small = document.createElement("canvas");
   211→      small.width = Math.round(source.w * scale);
   212→      small.height = Math.round(source.h * scale);
   213→      small.getContext("2d")!.drawImage(source.canvas, 0, 0, small.width, small.height);
   214→      const dataUrl = small.toDataURL("image/jpeg", 0.85);
   215→
   216→      const controller = new AbortController();
   217→      const timer = window.setTimeout(() => controller.abort(), 100_000);
   218→      const res = await fetch("/api/detect", {
   219→        method: "POST",
   220→        headers: { "Content-Type": "application/json" },
   221→        body: JSON.stringify({ image: dataUrl, width: source.w, height: source.h }),
   222→        signal: controller.signal,
   223→      });
   224→      window.clearTimeout(timer);
   225→      const json = (await res.json()) as {
   226→        ok: boolean;
   227→        boxes?: Array<{ x: number; y: number; w: number; h: number; label?: string }>;
   228→        error?: string;
   229→      };
   230→      if (!res.ok || !json.ok) {
   231→        throw new Error(json.error || `پاسخ سرور: ${res.status}`);
   232→      }
   233→      const mapped: Box[] = (json.boxes ?? []).map((b) => ({
   234→        id: makeBoxId(),
   235→        x: Math.round(b.x * source.w),
   236→        y: Math.round(b.y * source.h),
   237→        w: Math.round(b.w * source.w),
   238→        h: Math.round(b.h * source.h),
   239→        label: b.label,
   240→      }));
   241→      setBoxes(mapped);
   242→      setSelectedId(null);
   243→      if (mapped.length === 0) {
   244→        toast.warning("هوش مصنوعی کادری برنگرداند", {
   245→          description: "تشخیص محلی بینایی ماشین را امتحان کنید",
   246→        });
   247→      } else {
   248→        toast.success(`${mapped.length} کادر از Vision AI دریافت شد`);
   249→      }
   250→    } catch (err) {
   251→      // on static hosting (GitHub Pages) there is no /api/detect —
   252→      // fall back gracefully to the fully offline local vision engine
   253→      const msg =
   254→        err instanceof DOMException && err.name === "AbortError"
   255→          ? "زمان پاسخ سرور به پایان رسید"
   256→          : err instanceof Error
   257→            ? err.message
   258→            : "خطای ناشناخته";
   259→      toast.info("موتور AI سروری در دسترس نیست", {
   260→        description: `${msg} — تشخیص محلی روی دستگاه شما اجرا می‌شود`,
   261→      });
   262→      localDetect();
   263→    } finally {
   264→      setAiBusy(false);
   265→    }
   266→  }, [localDetect, source]);
   267→
   268→  const processCrop = useCallback(
   269→    (box: Box): { original: HTMLCanvasElement; processed: HTMLCanvasElement } => {
   270→      const original = cropToCanvas(source!.canvas, box);
   271→      if (!bgEnabled) {
   272→        return { original, processed: original };
   273→      }
   274→      const img = original.getContext("2d", { willReadFrequently: true })!.getImageData(
   275→        0,
   276→        0,
   277→        original.width,
   278→        original.height
   279→      );
   280→      const result = removeBackground(img, {
   281→        tolerance: bgTolerance,
   282→        edgeBarrier,
   283→        feather,
   284→        defringe,
   285→        removeHoles,
   286→        holeMaxPercent: holeCap,
   287→      });
   288→      const processed = document.createElement("canvas");
   289→      processed.width = original.width;
   290→      processed.height = original.height;
   291→      processed.getContext("2d")!.putImageData(result.image, 0, 0);
   292→      return { original, processed };
   293→    },
   294→    [bgEnabled, bgTolerance, defringe, edgeBarrier, feather, holeCap, removeHoles, source]
   295→  );
   296→
   297→  const openMaskEditor = useCallback(() => {
   298→    if (!source) return;
   299→    const box = boxes.find((b) => b.id === selectedId);
   300→    if (!box) {
   301→      toast.error("یک کادر را انتخاب کنید");
   302→      return;
   303→    }
   304→    const { original, processed } = processCrop(box);
   305→    const idx = boxes.findIndex((b) => b.id === selectedId);
   306→    setMaskKey((k) => k + 1);
   307→    setMaskState({
   308→      original,
   309→      processed,
   310→      name: frameName(`${prefix.trim() || "sprite"}_{i}`, idx, boxes.length),
   311→    });
   312→  }, [boxes, prefix, processCrop, selectedId, source]);
   313→
   314→  /** cut every box → add to animator → optionally download a transparent-WebP ZIP */
   315→  const cropAll = useCallback(async () => {
   316→    if (!source) return;
   317→    if (boxes.length === 0) {
   318→      toast.error("کادری برای برش وجود ندارد");
   319→      return;
   320→    }
   321→    setCropping(true);
   322→    try {
   323→      const inputs: Array<{ name: string; canvas: HTMLCanvasElement; source: "sheet" }> = [];
   324→      for (let i = 0; i < boxes.length; i++) {
   325→        const { processed } = processCrop(boxes[i]);
   326→        inputs.push({
   327→          name: frameName(`${prefix.trim() || "sprite"}_{i}`, i, boxes.length),
   328→          canvas: processed,
   329→          source: "sheet",
   330→        });
   331→        // keep UI responsive
   332→        await new Promise((r) => setTimeout(r, 0));
   333→      }
   334→      const n = addFrames(inputs);
   335→      if (autoDownload) {
   336→        toast.loading("در حال ساخت ZIP شفاف...", { id: "crop-zip" });
   337→        await downloadFramesZip(inputs, `${prefix.trim() || "sprite"}-cut`, 0.95);
   338→        toast.success(`${n} فایل برش‌شده دانلود شد`, {
   339→          id: "crop-zip",
   340→          description: bgEnabled
   341→            ? "ZIP با فریم‌های WebP شفاف — در انیماتور هم آماده است"
   342→            : "در تب انیماتور هم آماده است",
   343→        });
   344→      } else {
   345→        toast.success(`${n} فریم به انیماتور اضافه شد`, {
   346→          description: bgEnabled ? "با حذف پس‌زمینه و لبه نرم" : "برش مستقیم بدون حذف پس‌زمینه",
   347→        });
   348→      }
   349→    } catch (err) {
   350→      toast.error(err instanceof Error ? err.message : "خطا در برش فریم‌ها");
   351→    } finally {
   352→      setCropping(false);
   353→    }
   354→  }, [addFrames, autoDownload, bgEnabled, boxes, prefix, processCrop, source]);
   355→
   356→  /** download a single cut frame as transparent WebP */
   357→  const downloadOneBox = useCallback(
   358→    async (box: Box, index: number) => {
   359→      if (!source) return;
   360→      try {
   361→        const { processed } = processCrop(box);
   362→        const name = frameName(`${prefix.trim() || "sprite"}_{i}`, index, boxes.length);
   363→        const { blob, actualType } = await canvasToBlob(processed, "image/webp", 0.95);
   364→        downloadBlob(blob, `${name}.${extFor(actualType)}`);
   365→        toast.success(`«${name}.webp» دانلود شد`);
   366→      } catch (err) {
   367→        toast.error(err instanceof Error ? err.message : "دانلود فریم ناموفق بود");
   368→      }
   369→    },
   370→    [boxes.length, prefix, processCrop, source]
   371→  );
   372→
   373→  // graphical thumbnails for detected boxes (recomputed only when boxes settle)
   374→  const boxThumbs = useMemo(() => {
   375→    if (!source) return [];
   376→    return boxes.map((b, i) => {
   377→      const crop = cropToCanvas(source.canvas, b);
   378→      return {
   379→        id: b.id,
   380→        index: i,
   381→        thumb: makeThumb(crop, 132),
   382→        title: `${b.label ? `${b.label} — ` : ""}${Math.round(b.w)}×${Math.round(b.h)}`,
   383→      };
   384→    });
   385→  }, [boxes, source]);
   386→
   387→  const selectedBox = boxes.find((b) => b.id === selectedId) ?? null;
   388→
   389→  // ---------- empty state: graphical dropzone hero ----------
   390→  if (!source) {
   391→    return (
   392→      <div className="flex flex-col gap-4">
   393→        <input
   394→          ref={fileInputRef}
   395→          type="file"
   396→          accept="image/*"
   397→          className="hidden"
   398→          onChange={(e) => void onFiles(e.target.files)}
   399→        />
   400→        <div
   401→          className="hero-dots relative cursor-pointer overflow-hidden rounded-3xl border-2 border-dashed border-violet-line bg-surface p-8 text-center transition-colors hover:border-violet-line-strong sm:p-12"
   402→          onClick={() => fileInputRef.current?.click()}
   403→          onDragOver={(e) => e.preventDefault()}
   404→          onDrop={(e) => {
   405→            e.preventDefault();
   406→            void onFiles(e.dataTransfer.files);
   407→          }}
   408→        >
   409→          <div className="pointer-events-none absolute -top-28 right-1/2 h-72 w-72 translate-x-1/2 rounded-full bg-violet-tint blur-3xl" />
   410→          <div className="pointer-events-none absolute -bottom-28 left-1/4 h-56 w-56 rounded-full bg-cyan-tint blur-3xl" />
   411→          <div className="relative">
   412→            <div className="hero-icon-glow mx-auto grid h-20 w-20 place-items-center rounded-2xl metallic-gradient">
   413→              <ImageUp className="h-9 w-9 text-white" />
   414→            </div>
   415→            <h2 className="mt-5 text-xl font-black metallic-text sm:text-2xl">
   416→              شیت تصویر را اینجا رها کنید
   417→            </h2>
   418→            <p className="mx-auto mt-2 max-w-xl text-sm leading-6 text-soft">
   419→              آیتم‌ها خودکار شناسایی می‌شوند، پس‌زمینه حذف می‌شود و با یک کلیک، همه فایل‌های
   420→              برش‌خورده به‌صورت ZIP شفاف دانلود می‌شوند.
   421→            </p>
   422→            <div className="mt-6 flex flex-wrap items-center justify-center gap-3">
   423→              <Button
   424→                type="button"
   425→                onClick={(e) => {
   426→                  e.stopPropagation();
   427→                  fileInputRef.current?.click();
   428→                }}
   429→                className="metallic-gradient text-white font-bold glow-purple hover:brightness-110"
   430→              >
   431→                <ImageUp className="h-4 w-4" />
   432→                انتخاب تصویر
   433→              </Button>
   434→              <Button
   435→                type="button"
   436→                variant="secondary"
   437→                onClick={(e) => {
   438→                  e.stopPropagation();
   439→                  loadSample();
   440→                }}
   441→                className="border border-line bg-chip hover:border-cyan-line-strong"
   442→              >
   443→                <Sparkles className="h-4 w-4 text-cyan-ink" />
   444→                شیت نمونه تستی
   445→              </Button>
   446→            </div>
   447→            <p className="mt-3 text-xs text-faint" dir="ltr">
   448→              PNG · JPG · WebP — Drag &amp; Drop
   449→            </p>
   450→
   451→            <div className="mx-auto mt-9 grid max-w-3xl gap-3 sm:grid-cols-3">
   452→              {FEATURES.map((f) => (
   453→                <div
   454→                  key={f.title}
   455→                  className="rounded-2xl border border-line bg-surface p-4 text-right transition-colors hover:border-line-strong"
   456→                >
   457→                  <span className={cn("mb-2.5 grid h-9 w-9 place-items-center rounded-xl border", f.cls)}>
   458→                    <f.icon className="h-4.5 w-4.5" />
   459→                  </span>
   460→                  <p className="text-sm font-bold text-foreground">{f.title}</p>
   461→                  <p className="mt-1 text-xs leading-4 text-faint">{f.desc}</p>
   462→                </div>
   463→              ))}
   464→            </div>
   465→          </div>
   466→        </div>
   467→      </div>
   468→    );
   469→  }
   470→
   471→  // ---------- loaded state ----------
   472→  return (
   473→    <div className="flex flex-col gap-4">
   474→      <input
   475→        ref={fileInputRef}
   476→        type="file"
   477→        accept="image/*"
   478→        className="hidden"
   479→        onChange={(e) => void onFiles(e.target.files)}
   480→      />
   481→
   482→      <div className="grid gap-4 xl:grid-cols-[1fr_340px]">
   483→        {/* ---------- canvas column ---------- */}
   484→        <div className="flex flex-col gap-3">
   485→          {/* compact toolbar */}
   486→          <div className="flex flex-wrap items-center gap-2 rounded-2xl panel-glass p-2.5">
   487→            <Button
   488→              type="button"
   489→              size="sm"
   490→              variant="secondary"
   491→              onClick={() => fileInputRef.current?.click()}
   492→              className="border border-line bg-chip"
   493→            >
   494→              <ImageUp className="h-4 w-4" />
   495→              تعویض تصویر
   496→            </Button>
   497→            <Button
   498→              type="button"
   499→              size="sm"
   500→              variant="secondary"
   501→              disabled={aiBusy}
   502→              onClick={() => void aiDetect()}
   503→              className="border border-violet-line bg-violet-tint text-violet-ink hover:border-violet-line-strong"
   504→            >
   505→              {aiBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Cpu className="h-4 w-4" />}
   506→              تشخیص AI
   507→            </Button>
   508→            <Button
   509→              type="button"
   510→              size="sm"
   511→              variant="secondary"
   512→              disabled={localBusy}
   513→              onClick={localDetect}
   514→              className="border border-cyan-line bg-cyan-tint text-cyan-ink hover:border-cyan-line-strong"
   515→            >
   516→              {localBusy ? (
   517→                <Loader2 className="h-4 w-4 animate-spin" />
   518→              ) : (
   519→                <ScanLine className="h-4 w-4" />
   520→              )}
   521→              تشخیص محلی
